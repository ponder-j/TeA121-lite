cmake_minimum_required(VERSION 3.22)
project(tea121_extractor LANGUAGES C CXX)

option(TEA121_BUILD_LLVM_EXTRACTOR "Build the LLVM 15 MiniIR extractor" ON)
if(TEA121_BUILD_LLVM_EXTRACTOR)
  find_package(LLVM 15 CONFIG QUIET)
  if(LLVM_FOUND)
    message(STATUS "LLVM ${LLVM_PACKAGE_VERSION} found; building tea121-extract")
    include_directories(SYSTEM ${LLVM_INCLUDE_DIRS})
    add_definitions(${LLVM_DEFINITIONS})
    add_executable(tea121-extract llvm-extractor/extractor.cpp)
    set_target_properties(tea121-extract PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
    llvm_map_components_to_libnames(LLVM_LIBS Core IRReader Support)
    target_link_libraries(tea121-extract PRIVATE ${LLVM_LIBS})
  else()
    message(WARNING "LLVM 15 not found; Python MiniIR analyzer remains fully usable")
  endif()
endif()
